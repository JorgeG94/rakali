   pure subroutine qrange_three_diff_impl(dq0, dq1, dq2, qmin, qmax)
      !$acc routine seq
      real(wp), intent(in) :: dq0, dq1, dq2
      real(wp), intent(out) :: qmin, qmax

      qmax = max(dq0, dq0 + dq1, dq0 + dq2, 0.0_wp)
      qmin = min(dq0, dq0 + dq1, dq0 + dq2, 0.0_wp)
