   subroutine ensure_device(h)
      type(solver_handle_t), pointer, intent(inout) :: h

      if (.not. h%on_device) then
         call solver_enter_data(h%state)
         h%on_device = .true.
      end if
      h%host_is_current = .false.
   end subroutine ensure_device
