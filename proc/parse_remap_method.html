   pure integer function parse_remap_method(str) result(method)
      !! Convert a namelist string to a REMAP_* constant.
      character(len=*), intent(in) :: str
      select case (trim(adjustl(str)))
      case ("pcm")
         method = REMAP_PCM
      case ("plm")
         method = REMAP_PLM
      case ("ppm")
         method = REMAP_PPM
      case ("ppm_h4")
         method = REMAP_PPM_H4
      case ("pqm")
         method = REMAP_PQM
      case default
         method = REMAP_PLM
      end select
   end function parse_remap_method
