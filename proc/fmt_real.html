   pure function fmt_real(v) result(str)
      !! Format a real in a stable, namelist-valid exponential form.
      real(rk), intent(in) :: v
      character(len=:), allocatable :: str
      character(len=32) :: buf
      write (buf, "(E18.10)") v
      str = trim(adjustl(buf))
   end function fmt_real
