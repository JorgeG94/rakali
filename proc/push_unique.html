   subroutine push_unique(list, n, val)
      !! Append val to list(1:n) if not already present; bump n.
      integer, intent(inout) :: list(:)
      integer, intent(inout) :: n
      integer, intent(in) :: val
      integer :: s
      do s = 1, n
         if (list(s) == val) return
      end do
      n = n + 1
      list(n) = val
   end subroutine push_unique
