   subroutine nc_put_var_4d_slice(ncid, varid, data, time_index)
      !! Write a 3D slice into a 4D variable at a given time index

      integer, intent(in) :: varid
      real(wp), intent(in) :: data(:, :, :)

                    "writing 4D slice")
