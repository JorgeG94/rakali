   pure function bc_type_from_string(name) result(bc_type)
      !! Convert boundary type string to integer constant
      character(len=*), intent(in) :: name
      integer :: bc_type

      select case (trim(name))
      case ("wall")
         bc_type = BC_WALL
      case ("open")
         bc_type = BC_OPEN
      case ("tidal")
         bc_type = BC_TIDAL
      case ("nested")
         bc_type = BC_NESTED
      case ("inflow")
         bc_type = BC_INFLOW
      case ("discharge")
         bc_type = BC_DISCHARGE
      case ("clamped")
         bc_type = BC_CLAMPED
      case ("sponge")
         bc_type = BC_SPONGE
      case ("chapman")
         bc_type = BC_CHAPMAN
      case default
         bc_type = BC_WALL
      end select

   end function bc_type_from_string
